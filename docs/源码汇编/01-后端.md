package com.noteworkshop.admin;

import com.baomidou.mybatisplus.core.toolkit.Wrappers;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.noteworkshop.ai.AiCallLog;
import com.noteworkshop.ai.AiCallLogMapper;
import com.noteworkshop.comment.Comment;
import com.noteworkshop.comment.CommentMapper;
import com.noteworkshop.comment.CommentService;
import com.noteworkshop.common.BizException;
import com.noteworkshop.common.PageResult;
import com.noteworkshop.follow.Follow;
import com.noteworkshop.follow.FollowMapper;
import com.noteworkshop.note.Note;
import com.noteworkshop.note.NoteMapper;
import com.noteworkshop.note.NoteService;
import com.noteworkshop.user.User;
import com.noteworkshop.user.UserMapper;
import lombok.RequiredArgsConstructor;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.util.StringUtils;

import java.time.LocalDateTime;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

@Slf4j
@Service
@RequiredArgsConstructor
public class AdminService {

    private final UserMapper userMapper;
    private final NoteMapper noteMapper;
    private final CommentMapper commentMapper;
    private final CommentService commentService;
    private final FollowMapper followMapper;
    private final AiCallLogMapper aiCallLogMapper;

    public Map<String, Object> dashboard() {
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("userCount", userMapper.selectCount(null));
        map.put("noteCount", noteMapper.selectCount(null));
        map.put("publicNoteCount", noteMapper.selectCount(Wrappers.<Note>lambdaQuery()
                .eq(Note::getIsPublic, 1).eq(Note::getStatus, NoteService.STATUS_NORMAL)));
        map.put("pendingNoteCount", noteMapper.selectCount(Wrappers.<Note>lambdaQuery()
                .eq(Note::getStatus, NoteService.STATUS_PENDING)));
        map.put("commentCount", commentMapper.selectCount(Wrappers.<Comment>lambdaQuery()
                .eq(Comment::getDeleted, 0)));
        map.put("followCount", followMapper.selectCount(null));
        map.put("aiTotal", aiCallLogMapper.selectCount(null));
        map.put("aiSuccess", aiCallLogMapper.selectCount(Wrappers.<AiCallLog>lambdaQuery()
                .eq(AiCallLog::getSuccess, 1)));
        map.put("aiFail", aiCallLogMapper.selectCount(Wrappers.<AiCallLog>lambdaQuery()
                .eq(AiCallLog::getSuccess, 0)));
        return map;
    }

    public PageResult<User> users(long page, long size, String keyword) {
        var query = Wrappers.<User>lambdaQuery().orderByDesc(User::getCreatedAt);
        if (StringUtils.hasText(keyword)) {
            query.and(w -> w.like(User::getUsername, keyword).or().like(User::getNickname, keyword));
        }
        Page<User> p = userMapper.selectPage(new Page<>(page, size), query);
        p.getRecords().forEach(u -> u.setPassword(null));
        return PageResult.of(p);
    }

    @Transactional
    public void setUserStatus(Long userId, Integer status) {
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw new BizException(404, "用户不存在");
        }
        if ("ADMIN".equals(user.getRole()) && status != null && status == 0) {
            throw new BizException(400, "不能禁用管理员账号");
        }
        user.setStatus(status);
        userMapper.updateById(user);
        log.info("管理员修改用户状态 userId={} status={}", userId, status);
    }

    public PageResult<Note> notes(long page, long size, Integer status, String keyword) {
        var query = Wrappers.<Note>lambdaQuery().orderByDesc(Note::getCreatedAt);
        if (status != null) {
            query.eq(Note::getStatus, status);
        }
        if (StringUtils.hasText(keyword)) {
            query.and(w -> w.like(Note::getTitle, keyword).or().like(Note::getBookName, keyword));
        }
        return PageResult.of(noteMapper.selectPage(new Page<>(page, size), query));
    }

    @Transactional
    public void review(Long noteId, boolean approved, String reason) {
        Note note = noteMapper.selectById(noteId);
        if (note == null) {
            throw new BizException(404, "笔记不存在");
        }
        if (note.getStatus() == null || note.getStatus() != NoteService.STATUS_PENDING) {
            throw new BizException(400, "该笔记不在待审核状态");
        }
        note.setStatus(approved ? NoteService.STATUS_NORMAL : NoteService.STATUS_REJECTED);
        note.setRejectReason(approved ? null : (StringUtils.hasText(reason) ? reason : "内容不符合公开规范"));
        note.setReviewedAt(LocalDateTime.now());
        noteMapper.updateById(note);
        log.info("笔记审核 noteId={} approved={} reason={}", noteId, approved, note.getRejectReason());
    }

    @Transactional
    public void takeDown(Long noteId) {
        Note note = noteMapper.selectById(noteId);
        if (note == null) {
            throw new BizException(404, "笔记不存在");
        }
        note.setStatus(NoteService.STATUS_OFFLINE);
        noteMapper.updateById(note);
        log.info("管理员下架笔记 noteId={}", noteId);
    }

    public void deleteComment(Long commentId) {
        commentService.adminDelete(commentId);
    }

    public Map<String, Object> aiStats() {
        long total = aiCallLogMapper.selectCount(null);
        long success = aiCallLogMapper.selectCount(Wrappers.<AiCallLog>lambdaQuery()
                .eq(AiCallLog::getSuccess, 1));
        List<Map<String, Object>> rows = aiCallLogMapper.selectMaps(
                Wrappers.<AiCallLog>query()
                        .select("scene", "count(*) as total",
                                "sum(case when success = 1 then 1 else 0 end) as success")
                        .groupBy("scene"));
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("total", total);
        map.put("success", success);
        map.put("fail", total - success);
        map.put("byScene", rows);
        return map;
    }
}

package com.noteworkshop.ai;

import com.baomidou.mybatisplus.core.toolkit.Wrappers;

import com.noteworkshop.note.Note;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.util.StringUtils;

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

@Slf4j
@Service
@RequiredArgsConstructor
public class AiService {

    private static final String NOTE_SYSTEM_PROMPT =
            "你是「笔记工坊」的读书笔记助手，输出使用简体中文，只依据用户给出的笔记内容作答，不要编造。";

    private final ChatClient.Builder chatClientBuilder;
    private final AiProperties aiProperties;
    private final NoteAiMapper noteAiMapper;
    private final AiCallLogMapper callLogMapper;
    private final AiChatSessionMapper sessionMapper;
    private final AiChatMessageMapper messageMapper;
    private final NoteService noteService;

    private volatile ChatClient chatClient;

    private ChatClient client() {
        if (chatClient == null) {
            synchronized (this) {
                if (chatClient == null) {
                    chatClient = chatClientBuilder.build();
                }
            }
        }
        return chatClient;
    }

    public Map<String, Object> status() {
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("enabled", aiProperties.isEnabled());
        map.put("scenes", List.of("SUMMARY", "AUDIENCE", "SCORE", "CHAT"));
        map.put("note", "需要配置环境变量 AI_API_KEY / AI_BASE_URL / AI_MODEL 后才能真正调用大模型");
        return map;
    }

    public Map<String, Object> generate(Long userId, Long noteId, String type) {
        ensureEnabled();
        Note note = noteService.requireOwned(userId, noteId);
        String prompt = buildNotePrompt(note, type);
        long start = System.currentTimeMillis();
        try {
            String content = callAi(NOTE_SYSTEM_PROMPT, prompt);
            logCall(userId, type, true, null, System.currentTimeMillis() - start);
            Map<String, Object> map = new LinkedHashMap<>();
            map.put("type", type);
            map.put("content", content);
            map.put("saved", false);
            return map;
        } catch (Exception e) {
            logCall(userId, type, false, e.getMessage(), System.currentTimeMillis() - start);
            log.warn("AI 调用失败 type={} noteId={}", type, noteId, e);
            throw new BizException(503, "AI 服务暂不可用：" + brief(e.getMessage()));
        }
    }

    @Transactional
    public Map<String, Object> save(Long userId, Long noteId, String type, String content) {
        noteService.requireOwned(userId, noteId);
        NoteAi existing = noteAiMapper.selectOne(Wrappers.<NoteAi>lambdaQuery()
                .eq(NoteAi::getNoteId, noteId).last("limit 1"));

            NoteAi ai = new NoteAi();
            ai.setNoteId(noteId);
            ai.setType(type);
            ai.setContent(content);
            noteAiMapper.insert(ai);
        } else {
            existing.setType(type);
            existing.setContent(content);
            noteAiMapper.updateById(existing);
        }
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("type", type);
        map.put("content", content);
        map.put("saved", true);
        return map;
    }

        NoteAi ai = noteAiMapper.selectOne(Wrappers.<NoteAi>lambdaQuery()
                .eq(NoteAi::getNoteId, noteId).last("limit 1"));
        Map<String, Object> map = new LinkedHashMap<>();
        map.put("type", ai == null ? null : ai.getType());
        map.put("content", ai == null ? null : ai.getContent());
        map.put("savedAt", ai == null ? null : ai.getCreatedAt());

    /**
     * 不加事务：AI 调用失败时，用户消息与调用日志需要保留（日志用于统计失败次数）

    public Map<String, Object> chat(Long userId, ChatRequest request) {
        ensureEnabled();
        AiChatSession session;
        if (request.sessionId() == null) {
            session = new AiChatSession();
            session.setUserId(userId);
            session.setTitle(shortTitle(request.message()));
            sessionMapper.insert(session);
        } else {
            session = requireSession(userId, request.sessionId());
        }

        List<AiChatMessage> history = messageMapper.selectList(Wrappers.<AiChatMessage>lambdaQuery()
                .eq(AiChatMessage::getSessionId, session.getId())
                .orderByDesc(AiChatMessage::getId)
                .last("limit " + aiProperties.getWindowSize()));
        Collections.reverse(history);

        AiChatMessage userMessage = new AiChatMessage();
        userMessage.setSessionId(session.getId());
        userMessage.setRole("user");
        userMessage.setContent(request.message());
        messageMapper.insert(userMessage);

        String prompt = buildChatPrompt(history, request.message());
        long start = System.currentTimeMillis();
        String reply;
        try {
            reply = callAi(aiProperties.getSystemPrompt(), prompt);
            logCall(userId, "CHAT", true, null, System.currentTimeMillis() - start);
        } catch (Exception e) {
            logCall(userId, "CHAT", false, e.getMessage(), System.currentTimeMillis() - start);
            log.warn("AI 客服调用失败 sessionId={}", session.getId(), e);
            throw new BizException(503, "AI 服务暂不可用：" + brief(e.getMessage()));
        }

        AiChatMessage assistantMessage = new AiChatMessage();
        assistantMessage.setSessionId(session.getId());
        assistantMessage.setRole("assistant");
        assistantMessage.setContent(reply);
        messageMapper.insert(assistantMessage);

        session.setUpdatedAt(LocalDateTime.now());
        sessionMapper.updateById(session);

        Map<String, Object> map = new LinkedHashMap<>();
        map.put("sessionId", session.getId());
        map.put("title", session.getTitle());
        map.put("reply", reply);
        return map;
    }

    private AiChatSession requireSession(Long userId, Long sessionId) {
        AiChatSession session = sessionMapper.selectById(sessionId);
        if (session == null) {
            throw new BizException(404, "会话不存在");
        }
        if (!session.getUserId().equals(userId)) {
            throw new BizException(403, "无权访问他人的会话");
        }

    }

    private String callAi(String system, String user) {
        return client().prompt()
                .system(system)
                .user(user)
                .call()
                .content();
    }

    private String buildNotePrompt(Note note, String type) {
        String body = "书名：" + note.getBookName() + "\n标题：" + note.getTitle() + "\n正文：\n" + note.getContent();
        return switch (type) {
            case "SUMMARY" -> "请阅读下面的读书笔记，用不超过 200 字总结核心内容与要点，直接输出总结正文。\n\n" + body;
            case "AUDIENCE" -> "请根据下面的读书笔记，推荐 3 类最适合阅读它的人群，并分别说明理由。\n\n" + body;
            case "SCORE" -> "请从「内容完整性」「整体相关性」「思考深度」三个维度给下面的读书笔记打分（每项 10 分），"
                    + "给出总分（30 分制）和一句改进建议。\n\n" + body;
            default -> throw new BizException(400, "不支持的 AI 类型");
        };
    }

    private String buildChatPrompt(List<AiChatMessage> history, String message) {
        StringBuilder sb = new StringBuilder();
        if (!history.isEmpty()) {
            sb.append("以下是本会话最近的历史对话，请结合上下文回答最后一个问题：\n");
            for (AiChatMessage m : history) {
                sb.append("user".equals(m.getRole()) ? "用户：" : "助手：")
                        .append(m.getContent()).append('\n');
            }
            sb.append('\n');
        }
        sb.append("用户：").append(message);
        return sb.toString();
    }

    private void ensureEnabled() {
        if (!aiProperties.isEnabled()) {
            throw new BizException(503, "AI 功能未启用（设置 AI_ENABLED=true 并配置 AI_API_KEY）");
        }
    }

    private void logCall(Long userId, String scene, boolean success, String error, long latency) {
        try {
            AiCallLog log = new AiCallLog();
            log.setUserId(userId);
            log.setScene(scene);
            log.setSuccess(success ? 1 : 0);
            log.setErrorMsg(error == null ? null : brief(error));
            log.setLatencyMs((int) Math.min(latency, Integer.MAX_VALUE));
            callLogMapper.insert(log);
        } catch (Exception e) {
            log.warn("写入 AI 调用日志失败", e);
        }
    }

    private String brief(String text) {
        if (!StringUtils.hasText(text)) {
            return "未知错误";
        }
        return text.length() <= 240 ? text : text.substring(0, 240);
    }

    private String shortTitle(String message) {
        String t = message.replaceAll("\\s+", " ").trim();
        return t.length() <= 20 ? t : t.substring(0, 20);
    }

package com.noteworkshop.comment;

import com.baomidou.mybatisplus.core.toolkit.Wrappers;
import com.noteworkshop.comment.dto.CommentRequest;
import com.noteworkshop.comment.dto.CommentVO;
import com.noteworkshop.common.BizException;

import com.noteworkshop.user.User;
import com.noteworkshop.user.UserMapper;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.function.Function;
import java.util.stream.Collectors;

@Slf4j
@Service
@RequiredArgsConstructor
public class CommentService {

    private final CommentMapper commentMapper;
    private final NoteMapper noteMapper;
    private final UserMapper userMapper;

    public List<CommentVO> listByNote(Long noteId) {
        // 评论跟随笔记可见性：私有/待审核/已下架的笔记，评论不对外暴露
        Note note = noteMapper.selectById(noteId);
        if (note == null || note.getStatus() == null || note.getStatus() != 1
                || note.getIsPublic() == null || note.getIsPublic() != 1) {
            throw new BizException(404, "笔记不存在或未公开");
        }
        List<Comment> comments = commentMapper.selectList(Wrappers.<Comment>lambdaQuery()
                .eq(Comment::getNoteId, noteId)
                .eq(Comment::getDeleted, 0)
                .orderByAsc(Comment::getId));
        if (comments.isEmpty()) {
            return List.of();
        }
        Map<Long, User> users = userMapper.selectByIds(
                        comments.stream().map(Comment::getUserId).distinct().toList())
                .stream().collect(Collectors.toMap(User::getId, Function.identity()));

        Map<Long, CommentVO> voMap = new HashMap<>();
        for (Comment c : comments) {
            User u = users.get(c.getUserId());
            voMap.put(c.getId(), CommentVO.of(c.getId(), c.getNoteId(), c.getUserId(),
                    u == null ? "已注销用户" : u.getNickname(),
                    u == null ? null : u.getAvatar(),
                    c.getParentId(), c.getContent(), c.getCreatedAt()));
        }
        List<CommentVO> roots = new ArrayList<>();
        for (Comment c : comments) {
            CommentVO vo = voMap.get(c.getId());
            CommentVO parent = voMap.get(c.getParentId());
            if (c.getParentId() == null || c.getParentId() == 0 || parent == null) {
                roots.add(vo);
            } else {
                parent.children().add(vo);
            }
        }
        return roots;
    }

    @Transactional
    public CommentVO add(Long userId, Long noteId, CommentRequest request) {
        Note note = noteMapper.selectById(noteId);
        if (note == null || note.getStatus() == null || note.getStatus() != 1
                || note.getIsPublic() == null || note.getIsPublic() != 1) {
            throw new BizException(404, "笔记不存在或未公开");
        }
        Long parentId = request.parentId() == null ? 0L : request.parentId();
        if (parentId != 0) {
            Comment parent = commentMapper.selectById(parentId);
            if (parent == null || parent.getDeleted() == 1 || !parent.getNoteId().equals(noteId)) {
                throw new BizException(400, "回复的评论不存在");
            }
        }
        Comment comment = new Comment();
        comment.setNoteId(noteId);
        comment.setUserId(userId);
        comment.setParentId(parentId);
        comment.setContent(request.content());
        comment.setDeleted(0);
        commentMapper.insert(comment);
        noteMapper.update(null, Wrappers.<Note>lambdaUpdate()
                .eq(Note::getId, noteId)
                .setSql("comment_count = comment_count + 1"));

        User user = userMapper.selectById(userId);
        log.info("用户 {} 评论了笔记 {}", userId, noteId);
        return CommentVO.of(comment.getId(), noteId, userId,
                user == null ? "" : user.getNickname(),
                user == null ? null : user.getAvatar(),
                parentId, comment.getContent(), comment.getCreatedAt());
    }

    @Transactional
    public void delete(Long userId, Long commentId) {
        Comment comment = commentMapper.selectById(commentId);
        if (comment == null || comment.getDeleted() == 1) {
            throw new BizException(404, "评论不存在");
        }
        if (!comment.getUserId().equals(userId)) {
            throw new BizException(403, "只能删除自己的评论");
        }
        removeCascade(comment);
    }

    @Transactional
    public void adminDelete(Long commentId) {
        Comment comment = commentMapper.selectById(commentId);
        if (comment == null || comment.getDeleted() == 1) {
            throw new BizException(404, "评论不存在");
        }
        removeCascade(comment);
        log.info("管理员删除评论 {}", commentId);
    }

    /** 笔记被删除时，级联逻辑删除其全部评论 */
    @Transactional
    public void deleteByNote(Long noteId) {
        commentMapper.update(null, Wrappers.<Comment>lambdaUpdate()
                .eq(Comment::getNoteId, noteId)
                .eq(Comment::getDeleted, 0)
                .set(Comment::getDeleted, 1));
    }

    /** 逻辑删除该评论及其所有子孙评论 */
    private void removeCascade(Comment root) {
        List<Comment> all = commentMapper.selectList(Wrappers.<Comment>lambdaQuery()
                .eq(Comment::getNoteId, root.getNoteId())
                .eq(Comment::getDeleted, 0));
        Map<Long, List<Comment>> childrenMap = all.stream()
                .collect(Collectors.groupingBy(c -> c.getParentId() == null ? 0L : c.getParentId()));

        Set<Long> toDelete = new HashSet<>();
        Deque<Long> queue = new ArrayDeque<>();
        queue.add(root.getId());
        toDelete.add(root.getId());
        while (!queue.isEmpty()) {
            Long current = queue.poll();
            for (Comment child : childrenMap.getOrDefault(current, List.of())) {
                if (toDelete.add(child.getId())) {
                    queue.add(child.getId());
                }
            }
        }
        commentMapper.update(null, Wrappers.<Comment>lambdaUpdate()
                .in(Comment::getId, toDelete)
                .set(Comment::getDeleted, 1));
        noteMapper.update(null, Wrappers.<Note>lambdaUpdate()
                .eq(Note::getId, root.getNoteId())
                .setSql("comment_count = GREATEST(comment_count - " + toDelete.size() + ", 0)"));
    }
}

package com.noteworkshop.file;

import com.noteworkshop.common.BizException;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.util.Locale;
import java.util.Set;
import java.util.UUID;

@Slf4j
@Service
public class FileStorageService {

    private static final Set<String> ALLOWED = Set.of("jpg", "jpeg", "png", "gif", "webp");

    @Value("${app.upload-dir}")
    private String uploadDir;

    public String store(MultipartFile file) {
        if (file == null || file.isEmpty()) {
            throw new BizException("请选择要上传的文件");
        }
        String original = file.getOriginalFilename() == null ? "" : file.getOriginalFilename();
        String ext = original.contains(".")
                ? original.substring(original.lastIndexOf('.') + 1).toLowerCase(Locale.ROOT)
                : "";
        if (!ALLOWED.contains(ext)) {
            throw new BizException("只支持 " + ALLOWED + " 格式的图片");
        }
        try {
            Path base = Paths.get(uploadDir).toAbsolutePath().normalize();
            Files.createDirectories(base);
            String name = UUID.randomUUID().toString().replace("-", "") + "." + ext;
            Path target = base.resolve(name);
            try (var in = file.getInputStream()) {
                Files.copy(in, target, StandardCopyOption.REPLACE_EXISTING);
            }
            log.info("文件上传成功 original={} saved={} size={}", original, name, file.getSize());
            return "/uploads/" + name;
        } catch (IOException e) {
            throw new BizException(500, "文件保存失败");
        }
    }
}

package com.noteworkshop.follow;

import com.baomidou.mybatisplus.core.toolkit.Wrappers;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.noteworkshop.common.BizException;
import com.noteworkshop.common.PageResult;
import com.noteworkshop.follow.dto.AuthorVO;
import com.noteworkshop.note.Note;
import com.noteworkshop.note.NoteMapper;
import com.noteworkshop.user.User;
import com.noteworkshop.user.UserMapper;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.function.Function;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class FollowService {

    private final FollowMapper followMapper;
    private final UserMapper userMapper;
    private final NoteMapper noteMapper;

    @Transactional
    public Map<String, Object> follow(Long userId, Long targetId) {
        if (userId.equals(targetId)) {
            throw new BizException("不能关注自己");
        }
        User target = userMapper.selectById(targetId);
        if (target == null) {
            throw new BizException(404, "用户不存在");
        }
        Long exists = followMapper.selectCount(Wrappers.<Follow>lambdaQuery()
                .eq(Follow::getFollowerId, userId)
                .eq(Follow::getFolloweeId, targetId));
        if (exists == null || exists == 0) {
            Follow follow = new Follow();
            follow.setFollowerId(userId);
            follow.setFolloweeId(targetId);
            followMapper.insert(follow);
        }
        return Map.of("followed", true, "fansCount", fansCount(targetId));
    }

    @Transactional
    public Map<String, Object> unfollow(Long userId, Long targetId) {
        followMapper.delete(Wrappers.<Follow>lambdaQuery()
                .eq(Follow::getFollowerId, userId)
                .eq(Follow::getFolloweeId, targetId));
        return Map.of("followed", false, "fansCount", fansCount(targetId));
    }

    public long fansCount(Long userId) {
        return followMapper.selectCount(Wrappers.<Follow>lambdaQuery()
                .eq(Follow::getFolloweeId, userId));
    }

    public PageResult<User> followers(Long targetId, long page, long size) {
        Page<Follow> p = followMapper.selectPage(new Page<>(page, size),
                Wrappers.<Follow>lambdaQuery()
                        .eq(Follow::getFolloweeId, targetId)
                        .orderByDesc(Follow::getCreatedAt));
        return toUsers(p, Follow::getFollowerId, page, size);
    }

    public PageResult<User> following(Long userId, long page, long size) {
        Page<Follow> p = followMapper.selectPage(new Page<>(page, size),
                Wrappers.<Follow>lambdaQuery()
                        .eq(Follow::getFollowerId, userId)
                        .orderByDesc(Follow::getCreatedAt));
        return toUsers(p, Follow::getFolloweeId, page, size);
    }

    public AuthorVO profile(Long targetId, Long currentUserId) {
        User user = userMapper.selectById(targetId);
        if (user == null) {
            throw new BizException(404, "用户不存在");
        }
        long notes = noteMapper.selectCount(Wrappers.<Note>lambdaQuery()
                .eq(Note::getUserId, targetId)
                .eq(Note::getIsPublic, 1)
                .eq(Note::getStatus, 1));
        boolean followed = currentUserId != null && followMapper.selectCount(
                Wrappers.<Follow>lambdaQuery()
                        .eq(Follow::getFollowerId, currentUserId)
                        .eq(Follow::getFolloweeId, targetId)) > 0;
        return new AuthorVO(user.getId(), user.getNickname(), user.getAvatar(), user.getBio(),
                fansCount(targetId), notes, followed);
    }

    /** 热门作者：按粉丝数倒序 */
    public List<AuthorVO> hotAuthors(int limit) {
        List<Map<String, Object>> rows = followMapper.selectMaps(
                Wrappers.<Follow>query()
                        .select("followee_id", "count(*) as cnt")
                        .groupBy("followee_id")
                        .orderByDesc("cnt")
                        .last("limit " + Math.max(1, Math.min(limit, 50))));
        if (rows.isEmpty()) {
            return List.of();
        }
        List<Long> ids = rows.stream()
                .map(r -> ((Number) r.get("followee_id")).longValue())
                .toList();
        Map<Long, User> users = userMapper.selectByIds(ids).stream()
                .collect(Collectors.toMap(User::getId, Function.identity()));
        List<AuthorVO> result = new ArrayList<>();
        for (Map<String, Object> row : rows) {
            Long id = ((Number) row.get("followee_id")).longValue();
            User u = users.get(id);
            if (u == null) {
                continue;
            }
            result.add(new AuthorVO(u.getId(), u.getNickname(), u.getAvatar(), u.getBio(),
                    ((Number) row.get("cnt")).longValue(),
                    noteMapper.selectCount(Wrappers.<Note>lambdaQuery()
                            .eq(Note::getUserId, id)
                            .eq(Note::getIsPublic, 1)
                            .eq(Note::getStatus, 1)),
                    false));
        }
        return result;
    }

    private PageResult<User> toUsers(Page<Follow> p, Function<Follow, Long> idGetter, long page, long size) {
        if (p.getRecords().isEmpty()) {
            return PageResult.of(List.of(), p.getTotal(), page, size);
        }
        List<Long> ids = p.getRecords().stream().map(idGetter).toList();
        Map<Long, User> users = userMapper.selectByIds(ids).stream()
                .collect(Collectors.toMap(User::getId, Function.identity()));
        List<User> ordered = ids.stream()
                .map(users::get)
                .filter(java.util.Objects::nonNull)
                .peek(u -> u.setPassword(null))
                .toList();
        return PageResult.of(ordered, p.getTotal(), page, size);
    }
}

package com.noteworkshop.note;

            if (!visible) {

                        .eq(NoteLike::getNoteId, noteId)) > 0;

        note.setLiked(liked);

                .eq(NoteAi::getNoteId, noteId).last("limit 1"));

    public Map<String, Object> toggleLike(Long userId, Long noteId) {
        Note note = noteMapper.selectById(noteId);
        if (note == null) {
            throw new BizException(404, "笔记不存在");
        }
        if (note.getIsPublic() == null || note.getIsPublic() != 1
                || note.getStatus() == null || note.getStatus() != STATUS_NORMAL) {
            throw new BizException(400, "只能点赞公开且审核通过的笔记");
        }
        Long exists = likeMapper.selectCount(Wrappers.<NoteLike>lambdaQuery()
                .eq(NoteLike::getUserId, userId)
                .eq(NoteLike::getNoteId, noteId));
        boolean liked;
        if (exists > 0) {
            likeMapper.delete(Wrappers.<NoteLike>lambdaQuery()
                    .eq(NoteLike::getUserId, userId)
                    .eq(NoteLike::getNoteId, noteId));
            noteMapper.update(null, Wrappers.<Note>lambdaUpdate()
                    .eq(Note::getId, noteId)
                    .setSql("like_count = GREATEST(like_count - 1, 0)"));
            liked = false;
        } else {
            NoteLike like = new NoteLike();
            like.setUserId(userId);
            like.setNoteId(noteId);
            likeMapper.insert(like);
            noteMapper.update(null, Wrappers.<Note>lambdaUpdate()
                    .eq(Note::getId, noteId)
                    .setSql("like_count = like_count + 1"));
            liked = true;
        }
        Note latest = noteMapper.selectById(noteId);
        return Map.of("liked", liked, "likeCount", latest.getLikeCount());
    }

    public PageResult<Note> listLiked(Long userId, long page, long size) {
        IPage<NoteLike> likePage = likeMapper.selectPage(new Page<>(page, size),
                Wrappers.<NoteLike>lambdaQuery()
                        .eq(NoteLike::getUserId, userId)
                        .orderByDesc(NoteLike::getCreatedAt));
        if (CollectionUtils.isEmpty(likePage.getRecords())) {
            return PageResult.of(Collections.emptyList(), 0, page, size);
        }
        List<Long> noteIds = likePage.getRecords().stream().map(NoteLike::getNoteId).toList();
        Map<Long, Note> noteMap = noteMapper.selectByIds(noteIds).stream()
                .collect(Collectors.toMap(Note::getId, Function.identity()));
        List<Note> notes = noteIds.stream()
                .map(noteMap::get)
                .filter(java.util.Objects::nonNull)
                .toList();
        attachAuthors(notes);
        return PageResult.of(notes, likePage.getTotal(), page, size);
    }

    /** 供 AI 模块使用：校验笔记归属 */
    public Note requireOwned(Long userId, Long noteId) {
        Note note = noteMapper.selectById(noteId);
        if (note == null) {
            throw new BizException(404, "笔记不存在");
        }
        if (!note.getUserId().equals(userId)) {
            throw new BizException(403, "无权操作他人的笔记");
        }
        return note;
    }

    private void saveImages(Long noteId, List<String> urls) {
        if (CollectionUtils.isEmpty(urls)) {
            return;
        }
        int sort = 0;
        for (String url : urls) {
            NoteImage image = new NoteImage();
            image.setNoteId(noteId);
            image.setUrl(url);
            image.setSort(sort++);
            imageMapper.insert(image);
        }
    }

    private List<String> imageUrls(Long noteId) {
        return imageMapper.selectList(Wrappers.<NoteImage>lambdaQuery()
                        .eq(NoteImage::getNoteId, noteId)
                        .orderByAsc(NoteImage::getSort))
                .stream().map(NoteImage::getUrl).toList();
    }

    private void attachAuthors(List<Note> notes) {
        if (CollectionUtils.isEmpty(notes)) {
            return;
        }
        List<Long> userIds = notes.stream().map(Note::getUserId).distinct().toList();
        Map<Long, User> users = userMapper.selectByIds(userIds).stream()
                .collect(Collectors.toMap(User::getId, Function.identity()));
        for (Note note : notes) {
            User author = users.get(note.getUserId());
            if (author != null) {
                note.setAuthorName(author.getNickname());
                note.setAuthorAvatar(author.getAvatar());
            }
        }
    }

    private void attachAuthor(Note note) {
        User author = userMapper.selectById(note.getUserId());
        if (author != null) {
            note.setAuthorName(author.getNickname());
            note.setAuthorAvatar(author.getAvatar());
        }
    }

package com.noteworkshop.security;

import com.noteworkshop.config.JwtProperties;
import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.time.Duration;
import java.util.Date;

@Component
public class JwtUtil {

    private final SecretKey key;
    private final long expireMillis;

    public JwtUtil(JwtProperties properties) {
        this.key = Keys.hmacShaKeyFor(properties.getSecret().getBytes(StandardCharsets.UTF_8));
        this.expireMillis = Duration.ofDays(properties.getExpireDays()).toMillis();
    }

    public String issue(Long userId, String username) {
        Date now = new Date();
        return Jwts.builder()
                .subject(String.valueOf(userId))
                .claim("username", username)
                .issuedAt(now)
                .expiration(new Date(now.getTime() + expireMillis))
                .signWith(key)
                .compact();
    }

    public Claims parse(String token) {
        return Jwts.parser()
                .verifyWith(key)
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }

    public Long userId(String token) {
        return Long.valueOf(parse(token).getSubject());
    }
}

package com.noteworkshop.user;

import com.baomidou.mybatisplus.core.toolkit.Wrappers;
import com.noteworkshop.common.BizException;
import com.noteworkshop.security.JwtUtil;
import com.noteworkshop.user.dto.AuthResponse;
import com.noteworkshop.user.dto.LoginRequest;
import com.noteworkshop.user.dto.ProfileRequest;
import com.noteworkshop.user.dto.RegisterRequest;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.util.StringUtils;

@Slf4j
@Service
@RequiredArgsConstructor
public class UserService {

    private final UserMapper userMapper;
    private final JwtUtil jwtUtil;
    private final BCryptPasswordEncoder passwordEncoder = new BCryptPasswordEncoder();

    @Transactional
    public User register(RegisterRequest request) {
        Long count = userMapper.selectCount(
                Wrappers.<User>lambdaQuery().eq(User::getUsername, request.username()));
        if (count != null && count > 0) {
            throw new BizException("用户名已存在");
        }
        User user = new User();
        user.setUsername(request.username());
        user.setPassword(passwordEncoder.encode(request.password()));
        user.setNickname(StringUtils.hasText(request.nickname()) ? request.nickname() : request.username());
        user.setRole("USER");
        user.setStatus(1);
        userMapper.insert(user);
        user.setPassword(null);
        log.info("用户注册成功 username={} id={}", user.getUsername(), user.getId());
        return user;
    }

    public AuthResponse login(LoginRequest request) {
        User user = userMapper.selectOne(
                Wrappers.<User>lambdaQuery().eq(User::getUsername, request.username()));
        if (user == null || !passwordEncoder.matches(request.password(), user.getPassword())) {
            throw new BizException(401, "用户名或密码错误");
        }
        if (user.getStatus() != null && user.getStatus() == 0) {
            throw new BizException(403, "账号已被禁用");
        }
        String token = jwtUtil.issue(user.getId(), user.getUsername());
        user.setPassword(null);
        log.info("用户登录成功 username={} id={}", user.getUsername(), user.getId());
        return new AuthResponse(token, user);
    }

    public User me(Long userId) {
        User user = userMapper.selectById(userId);
        if (user == null) {
            throw new BizException(404, "用户不存在");
        }
        user.setPassword(null);
        return user;
    }

    @Transactional
    public User updateProfile(Long userId, ProfileRequest request) {
        User user = me(userId);
        if (StringUtils.hasText(request.nickname())) {
            user.setNickname(request.nickname());
        }
        if (request.bio() != null) {
            user.setBio(request.bio());
        }
        if (StringUtils.hasText(request.avatar())) {
            user.setAvatar(request.avatar());
        }
        user.setPassword(null);
        userMapper.updateById(user);
        return user;
    }
}
