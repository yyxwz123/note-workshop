### frontend/src/api/index.js

```javascript
import axios from 'axios'

const http = axios.create({
  baseURL: '',
  timeout: 15000
})

let redirectingToLogin = false

function clearAuthStorage() {
  localStorage.removeItem('nw_token')
  localStorage.removeItem('nw_user')
}

function redirectToLogin() {
  clearAuthStorage()
  if (typeof window === 'undefined') return
  if (window.location.pathname === '/login' || redirectingToLogin) return
  redirectingToLogin = true
  window.location.assign('/login')
}

http.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('nw_token')
    if (token) {
      config.headers = config.headers || {}
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => Promise.reject(error)
)

http.interceptors.response.use(
  (response) => {
    const body = response.data
    if (body && typeof body === 'object' && typeof body.code !== 'undefined') {
      if (body.code === 0) {
        return body.data
      }
      const error = new Error(body.message || '请求失败，请稍后重试')
      error.code = body.code
      error.status = response.status
      if (body.code === 401) {
        redirectToLogin()
      }
      return Promise.reject(error)
    }
    return body
  },
  (error) => {
    if (error.response) {
      const body = error.response.data
      const code =
        body && typeof body.code !== 'undefined' ? body.code : error.response.status
      const message =
        (body && body.message) || `请求失败（HTTP ${error.response.status}）`
      const wrapped = new Error(message)
      wrapped.code = code
      wrapped.status = error.response.status
      if (code === 401 || error.response.status === 401) {
        redirectToLogin()
      }
      return Promise.reject(wrapped)
    }
    if (error.code === 'ECONNABORTED') {
      const timeoutError = new Error('请求超时，请稍后重试')
      timeoutError.code = 'TIMEOUT'
      return Promise.reject(timeoutError)
    }
    const networkError = new Error('网络异常，请确认后端服务已启动')
    networkError.code = 'NETWORK'
    return Promise.reject(networkError)
  }
)

export function uploadFile(file) {
  const formData = new FormData()
  formData.append('file', file)
  return http.post('/api/files/upload', formData)
}

export default http
```

```vue
import { computed, onMounted, ref } from 'vue'

  name: 'AdminAiStats',

    const stats = ref({})
    const loading = ref(true)
    const error = ref('')

    const rows = computed(() => (Array.isArray(stats.value.byScene) ? stats.value.byScene : []))

    const successRate = computed(() => {
      const total = Number(stats.value.total) || 0
      const success = Number(stats.value.success) || 0
      if (!total) return 0
      return Math.round((success / total) * 100)
    })

    function formatNumber(value) {

    function sceneLabel(scene) {

    }

    function rowFail(row) {
      return Math.max(0, (Number(row.total) || 0) - (Number(row.success) || 0))
    }

    function rowRate(row) {
      const total = Number(row.total) || 0
      if (!total) return 0
      return Math.round(((Number(row.success) || 0) / total) * 100)
    }

    async function load() {
      loading.value = true
      error.value = ''

        stats.value = await api.get('/api/admin/ai/stats')
      } catch (err) {
        error.value = err.message || 'AI 统计加载失败'
        notifyError(err)
      } finally {
        loading.value = false
      }
    }

    onMounted(load)

    return {
      stats,
      loading,
      error,
      rows,
      successRate,
      formatNumber,
      sceneLabel,
      rowFail,
      rowRate,
      load
    }
  }
}
</script>
```

```vue
<template>
  <nav v-if="totalPages > 1" class="pagination" aria-label="分页">
    <button
      class="pagination__btn"
      type="button"
      :disabled="page <= 1 || disabled"
      @click="go(page - 1)"
    >
      上一页
    </button>
    <template v-for="(item, index) in pages" :key="`${item}-${index}`">
      <span v-if="item === '...'" class="pagination__ellipsis">…</span>
      <button
        v-else
        class="pagination__btn"
        :class="{ 'is-active': item === page }"
        type="button"
        :disabled="disabled"
        @click="go(item)"
      >
        {{ item }}
      </button>
    </template>
    <button
      class="pagination__btn"
      type="button"
      :disabled="page >= totalPages || disabled"
      @click="go(page + 1)"
    >
      下一页
    </button>
    <span class="pagination__total">共 {{ total }} 条</span>
  </nav>
</template>

<script>
import { computed } from 'vue'

export default {
  name: 'AppPagination',
  props: {
    page: { type: Number, default: 1 },
    size: { type: Number, default: 10 },
    total: { type: Number, default: 0 },
    disabled: { type: Boolean, default: false }
  },
  emits: ['change'],
  setup(props, { emit }) {
    const totalPages = computed(() =>
      Math.max(1, Math.ceil((Number(props.total) || 0) / (Number(props.size) || 1)))
    )

    const pages = computed(() => {
      const last = totalPages.value
      if (last <= 7) {
        return Array.from({ length: last }, (_, index) => index + 1)
      }
      const list = [1]
      const start = Math.max(2, props.page - 1)
      const end = Math.min(last - 1, props.page + 1)
      if (start > 2) list.push('...')
      for (let i = start; i <= end; i += 1) list.push(i)
      if (end < last - 1) list.push('...')
      list.push(last)
      return list
    })

    function go(target) {
      if (props.disabled) return
      if (target < 1 || target > totalPages.value || target === props.page) return
      emit('change', target)
    }

    return { totalPages, pages, go }
  }
}
</script>
```

```javascript
import { reactive } from 'vue'
import api from '@/api'

function readStoredUser() {
  try {
    const raw = localStorage.getItem('nw_user')
    return raw ? JSON.parse(raw) : null
  } catch (error) {
    localStorage.removeItem('nw_user')
    return null
  }
}

function sanitizeUser(user) {
  if (!user) return null
  const copy = { ...user }
  delete copy.password
  return copy
}

const state = reactive({
  token: localStorage.getItem('nw_token') || '',
  user: sanitizeUser(readStoredUser())
})

function persist(token, user) {
  state.token = token || ''
  state.user = sanitizeUser(user)
  if (state.token) {
    localStorage.setItem('nw_token', state.token)
  } else {
    localStorage.removeItem('nw_token')
  }
  if (state.user) {
    localStorage.setItem('nw_user', JSON.stringify(state.user))
  } else {
    localStorage.removeItem('nw_user')
  }
}

function isLoggedIn() {
  return Boolean(state.token && state.user)
}

function isAdmin() {
  return Boolean(state.user && state.user.role === 'ADMIN')
}

async function login(username, password) {
  const data = await api.post('/api/auth/login', { username, password })
  if (!data || !data.token) {
    throw new Error('登录响应异常，请稍后重试')
  }
  persist(data.token, data.user)
  return state.user
}

function logout() {
  persist('', null)
}

function setUser(user) {
  state.user = sanitizeUser(user)
  if (state.user) {
    localStorage.setItem('nw_user', JSON.stringify(state.user))
  } else {
    localStorage.removeItem('nw_user')
  }
}

async function bootstrap() {
  if (!state.token) return
  try {
    const user = await api.get('/api/users/me')
    if (user) setUser(user)
  } catch (error) {
    // 401 已由 axios 拦截器统一清除登录态并跳转；其他错误保留本地登录态。
  }
}

export default {
  state,
  login,
  logout,
  setUser,
  isLoggedIn,
  isAdmin,
  bootstrap
}
```

```javascript
import { reactive } from 'vue'

export const toasts = reactive([])

let seed = 0

export function notify(message, type = 'info', duration = 3200) {
  const id = ++seed
  toasts.push({ id, message: String(message || ''), type })
  if (duration > 0) {
    window.setTimeout(() => removeToast(id), duration)
  }
  return id
}

export function notifySuccess(message) {
  return notify(message, 'success')
}

export function notifyError(error) {
  const message =
    typeof error === 'string'
      ? error
      : (error && error.message) || '操作失败，请稍后重试'
  return notify(message, 'error', 4200)
}

export function removeToast(id) {
  const index = toasts.findIndex((item) => item.id === id)
  if (index !== -1) toasts.splice(index, 1)
}
```

```javascript
export const NOTE_STATUS = {

  0: { label: '已下架', tone: 'muted', hint: '笔记已被管理员下架' }

export const AI_TYPE_LABEL = {
  SUMMARY: '内容总结',
  AUDIENCE: '适合人群',
  SCORE: '质量评分'
}

export function noteStatusMeta(status) {
  return NOTE_STATUS[status] || { label: '未知', tone: 'muted', hint: '' }
}

export function aiTypeLabel(type) {
  if (!type) return 'AI 辅助'
  return AI_TYPE_LABEL[type] || type
}

function pad(value) {
  return String(value).padStart(2, '0')
}

export function formatDate(value) {
  if (!value) return ''
  const date = new Date(value)
  if (Number.isNaN(date.getTime())) return String(value)
  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())} ${pad(
    date.getHours()
  )}:${pad(date.getMinutes())}`
}

export function timeAgo(value) {
  if (!value) return ''
  const date = new Date(value)
  if (Number.isNaN(date.getTime())) return String(value)
  const diff = Date.now() - date.getTime()
  if (diff < 0) return formatDate(value)
  if (diff < 60 * 1000) return '刚刚'
  if (diff < 60 * 60 * 1000) return `${Math.floor(diff / 60000)} 分钟前`
  if (diff < 24 * 60 * 60 * 1000) return `${Math.floor(diff / 3600000)} 小时前`
  if (diff < 30 * 24 * 60 * 60 * 1000) return `${Math.floor(diff / 86400000)} 天前`
  return formatDate(value)
}

export function excerpt(text, length = 110) {
  const value = String(text || '').replace(/\s+/g, ' ').trim()
  if (!value) return '这篇笔记还没有正文。'
  return value.length > length ? `${value.slice(0, length)}…` : value
}

export function mediaUrl(url) {
  const value = String(url || '').trim()
  if (!value) return ''
  if (/^(https?:)?\/\//i.test(value) || value.startsWith('data:') || value.startsWith('blob:')) {
    return value
  }
  return value.startsWith('/') ? value : `/${value}`
}

export function displayName(user) {
  if (!user) return '匿名用户'
  return user.nickname || user.username || '匿名用户'
}

export function countComments(list) {
  if (!Array.isArray(list)) return 0
  return list.reduce((sum, item) => sum + 1 + countComments(item.children), 0)
}

export function flattenReplies(children) {
  const result = []
  const walk = (items) => {
    if (!Array.isArray(items)) return
    items.forEach((item) => {
      result.push(item)
      walk(item.children)
    })
  }
  walk(children)
  return result
}
```

### frontend/src/utils/hydrate.js

```javascript
import api from '@/api'

/**
 * 列表接口不返回配图，这里仅在缺少 images 时逐个补取详情，用于展示首图。
 * 失败时保持占位图，不影响列表主流程。
 */
export async function hydrateNoteCovers(notes, isStale = () => false) {
  if (!Array.isArray(notes)) return
  const missing = notes.filter(
    (note) => note && note.id && (!Array.isArray(note.images) || note.images.length === 0)
  )
  if (!missing.length) return

  await Promise.allSettled(
    missing.map(async (note) => {
      try {
        const detail = await api.get(`/api/public/notes/${note.id}`)
        if (!isStale() && detail) {
          note.images = Array.isArray(detail.images) ? detail.images : []
        }
      } catch (error) {
        if (!isStale()) note.images = []
      }
    })
  )
}
```

```vue
<script>
import { computed, nextTick, onMounted, ref, watch } from 'vue'
import api from '@/api'
import auth from '@/store/auth'
import { notifyError, notifySuccess } from '@/store/toast'
import UserAvatar from '@/components/UserAvatar.vue'
import EmptyState from '@/components/EmptyState.vue'
import { aiTypeLabel, formatDate, timeAgo } from '@/utils/format'

const AI_TYPES = [
  { value: 'SUMMARY', label: '内容总结', icon: '✦', desc: '提炼笔记的核心观点与要点' },
  { value: 'AUDIENCE', label: '适合人群', icon: '◎', desc: '推荐 3 类最适合阅读的人群' },
  { value: 'SCORE', label: '质量评分', icon: '★', desc: '从三个维度评分并给出建议' }
]

const SUGGESTED_QUESTIONS = [
  '怎么发表一篇公开笔记？',
  '公开笔记审核需要多久？',
  '如何让笔记更容易被推荐？'
]

export default {
  name: 'AiView',
  components: { UserAvatar, EmptyState },
  setup() {
    const activeTab = ref('read')
    const currentUser = computed(() => auth.state.user || {})

    const aiStatus = ref(null)
    const statusLoading = ref(false)
    const statusError = ref('')
    const aiUnavailable = ref(false)
    const aiUnavailableMessage = ref('')

    const myNotes = ref([])
    const notesLoading = ref(true)
    const notesError = ref('')
    const selectedNoteId = ref(null)
    const selectedType = ref('SUMMARY')
    const existingAi = ref(null)
    const generating = ref(false)
    const result = ref(null)
    const readError = ref('')
    const savingResult = ref(false)

    const sessions = ref([])
    const sessionsLoading = ref(false)
    const sessionsError = ref('')
    const sessionsLoaded = ref(false)
    const currentSessionId = ref(null)
    const messages = ref([])
    const messagesLoading = ref(false)
    const chatInput = ref('')
    const sending = ref(false)
    const chatError = ref('')
    const chatBody = ref(null)

    const aiTypes = AI_TYPES
    const suggestedQuestions = SUGGESTED_QUESTIONS

    const statusPillClass = computed(() => {
      if (statusLoading.value) return 'is-unknown'
      if (aiUnavailable.value) return 'is-disabled'
      if (!aiStatus.value) return 'is-unknown'
      return aiStatus.value.enabled ? 'is-enabled' : 'is-disabled'
    })

    const statusLabel = computed(() => {
      if (statusLoading.value) return '检测中…'
      if (aiUnavailable.value) return 'AI 不可用'
      if (!aiStatus.value) return '状态未知'
      return aiStatus.value.enabled ? 'AI 已启用' : 'AI 未启用'
    })

    const scenes = computed(() => {
      if (aiStatus.value && Array.isArray(aiStatus.value.scenes)) {
        return aiStatus.value.scenes
      }
      return ['SUMMARY', 'AUDIENCE', 'SCORE', 'CHAT']
    })

    const statusNote = computed(() => (aiStatus.value && aiStatus.value.note) || '')

    const showBanner = computed(
      () => aiUnavailable.value || Boolean(aiStatus.value && aiStatus.value.enabled === false)
    )

    const bannerDetail = computed(() => {
      if (aiUnavailableMessage.value) return aiUnavailableMessage.value
      if (aiStatus.value && aiStatus.value.note) return aiStatus.value.note
      return '请在后端设置环境变量 AI_API_KEY 后重启服务，再回来重试。'
    })

    const currentSession = computed(
      () => sessions.value.find((item) => item.id === currentSessionId.value) || null
    )

    async function loadAiStatus() {
      statusLoading.value = true
      statusError.value = ''
      try {
        aiStatus.value = await api.get('/api/ai/status')
        if (aiStatus.value && aiStatus.value.enabled) {
          aiUnavailable.value = false
          aiUnavailableMessage.value = ''
        }
      } catch (err) {
        statusError.value = err.message || 'AI 状态加载失败'
      } finally {
        statusLoading.value = false
      }
    }

    async function loadMyNotes() {
      notesLoading.value = true
      notesError.value = ''
      try {
        const data = await api.get('/api/notes/mine', { params: { page: 1, size: 100 } })
        myNotes.value = Array.isArray(data.records) ? data.records : []
        if (myNotes.value.length) {
          const stillExists = myNotes.value.some((note) => note.id === selectedNoteId.value)
          if (!stillExists) {
            selectedNoteId.value = myNotes.value[0].id
          }
          await loadExistingAi()
        } else {
          selectedNoteId.value = null
          existingAi.value = null
        }
      } catch (err) {
        notesError.value = err.message || '我的笔记加载失败'
        notifyError(err)
      } finally {
        notesLoading.value = false
      }
    }

    async function loadExistingAi() {
      if (!selectedNoteId.value) {
        existingAi.value = null
        return
      }
      try {
        const data = await api.get(`/api/ai/notes/${selectedNoteId.value}`)
        existingAi.value = data && data.content ? data : null
      } catch (err) {
        existingAi.value = null
      }
    }

    watch(selectedNoteId, async () => {
      result.value = null
      readError.value = ''
      await loadExistingAi()
    })

    function chooseType(type) {
      selectedType.value = type
      result.value = null
      readError.value = ''
    }

    function friendlyAiError(err) {
      const message = (err && err.message) || 'AI 服务暂不可用，请稍后重试'
      const unavailable =
        (err && err.code === 503) ||
        /AI 未配置|AI 服务|AI 功能未启用|API_KEY|未配置/i.test(message)
      if (unavailable) {
        aiUnavailable.value = true
        aiUnavailableMessage.value =
          '后端 AI 服务暂不可用，请检查 AI_API_KEY、AI_BASE_URL 与 AI_MODEL 配置后重启服务。'
        return 'AI 未配置，请设置 AI_API_KEY 后重试。'
      }
      return message
    }

    async function generate() {
      if (!selectedNoteId.value) {
        notifyError('请先选择一篇笔记')
        return
      }
      generating.value = true
      readError.value = ''
      result.value = null
      try {
        const data = await api.post(`/api/ai/notes/${selectedNoteId.value}/generate`, {
          type: selectedType.value
        })
        result.value = data
        aiUnavailable.value = false
        notifySuccess('AI 内容生成完成')
      } catch (err) {
        readError.value = friendlyAiError(err)
        notifyError(err)
      } finally {
        generating.value = false
      }
    }

    async function saveResult() {
      if (!result.value || !selectedNoteId.value) return
      savingResult.value = true
      try {
        const data = await api.post(`/api/ai/notes/${selectedNoteId.value}/save`, {
          type: result.value.type,
          content: result.value.content
        })
        result.value.saved = true
        existingAi.value = { type: data.type, content: data.content }
        notifySuccess('已保存到笔记，详情页将展示 AI 辅助区块')
      } catch (err) {
        notifyError(err)
      } finally {
        savingResult.value = false
      }
    }

    async function copyResult() {
      if (!result.value) return
      try {
        await navigator.clipboard.writeText(result.value.content)
        notifySuccess('已复制到剪贴板')
      } catch (err) {
        notifyError('复制失败，请手动选择文本复制')
      }
    }

    function switchTab(tab) {
      activeTab.value = tab
      if (tab === 'chat' && !sessionsLoaded.value) {
        loadSessions()
      }
      if (tab === 'chat') {
        scrollChat()
      }
    }

    async function loadSessions() {
      sessionsLoading.value = true
      sessionsError.value = ''
      try {
        const data = await api.get('/api/ai/sessions')
        sessions.value = Array.isArray(data) ? data : []
        sessionsLoaded.value = true
      } catch (err) {
        sessionsError.value = err.message || '会话列表加载失败'
      } finally {
        sessionsLoading.value = false
      }
    }

    function newSession() {
      if (sending.value) {
        notifyError('请等待当前回复完成')
        return
      }
      currentSessionId.value = null
      messages.value = []
      chatInput.value = ''
      chatError.value = ''
      scrollChat()
    }

    async function openSession(session) {
      if (sending.value) {
        notifyError('请等待当前回复完成')
        return
      }
      currentSessionId.value = session.id
      chatError.value = ''
      messagesLoading.value = true
      try {
        const data = await api.get(`/api/ai/sessions/${session.id}/messages`)
        messages.value = Array.isArray(data) ? data : []
        scrollChat()
      } catch (err) {
        chatError.value = err.message || '会话记录加载失败'
        messages.value = []
      } finally {
        messagesLoading.value = false
      }
    }

    async function removeSession(session) {
      if (!session) return
      if (sending.value) {
        notifyError('请等待当前回复完成')
        return
      }
      if (!window.confirm(`确定删除会话「${session.title || `会话 #${session.id}`}」吗？`)) return
      try {
        await api.delete(`/api/ai/sessions/${session.id}`)
        if (currentSessionId.value === session.id) {
          currentSessionId.value = null
          messages.value = []
        }
        notifySuccess('会话已删除')
        await loadSessions()
      } catch (err) {
        notifyError(err)
      }
    }

    function quickAsk(question) {
      chatInput.value = question
      sendMessage()
    }

    async function sendMessage() {
      const text = chatInput.value.trim()
      if (!text || sending.value) return
      if (text.length > 2000) {
        notifyError('消息不能超过 2000 个字符')
        return
      }

      const sessionId = currentSessionId.value
      messages.value.push({
        id: `temp-user-${Date.now()}`,
        role: 'user',
        content: text,
        createdAt: new Date().toISOString()
      })
      chatInput.value = ''
      chatError.value = ''
      sending.value = true
      scrollChat()

      try {
        const data = await api.post('/api/ai/chat', { sessionId, message: text })
        currentSessionId.value = data.sessionId
        messages.value.push({
          id: `assistant-${Date.now()}`,
          role: 'assistant',
          content: data.reply,
          createdAt: new Date().toISOString()
        })
        aiUnavailable.value = false
        sessionsLoaded.value = false
        await loadSessions()
      } catch (err) {
        chatError.value = friendlyAiError(err)
        notifyError(err)
        sessionsLoaded.value = false
        loadSessions()
      } finally {
        sending.value = false
        scrollChat()
      }
    }

    function scrollChat() {
      nextTick(() => {
        const element = chatBody.value
        if (element) element.scrollTop = element.scrollHeight
      })
    }

    onMounted(() => {
      loadAiStatus()
      loadMyNotes()
    })

    return {
```

```vue
<script>
import { computed, onMounted, ref } from 'vue'
import api from '@/api'
import auth from '@/store/auth'
import { notifyError } from '@/store/toast'
import NoteCard from '@/components/NoteCard.vue'
import AppPagination from '@/components/AppPagination.vue'
import EmptyState from '@/components/EmptyState.vue'
import UserAvatar from '@/components/UserAvatar.vue'
import { hydrateNoteCovers } from '@/utils/hydrate'

const PAGE_SIZE = 9

export default {
  name: 'HomeView',
  components: { NoteCard, AppPagination, EmptyState, UserAvatar },
  setup() {
    const notes = ref([])
    const total = ref(0)
    const page = ref(1)
    const sort = ref('latest')
    const loading = ref(true)
    const error = ref('')

    const authors = ref([])
    const authorsLoading = ref(true)
    const authorsError = ref('')

    const loggedIn = computed(() => auth.isLoggedIn())
    const size = PAGE_SIZE
    let requestSeed = 0

    async function loadNotes() {
      const seed = ++requestSeed
      loading.value = true
      error.value = ''
      try {
        const data = await api.get('/api/public/notes', {
          params: { page: page.value, size: PAGE_SIZE, sort: sort.value }
        })
        if (seed !== requestSeed) return
        notes.value = Array.isArray(data.records) ? data.records : []
        total.value = Number(data.total) || 0
        hydrateNoteCovers(notes.value, () => seed !== requestSeed)
      } catch (err) {
        if (seed !== requestSeed) return
        error.value = err.message || '笔记列表加载失败'
        notes.value = []
        total.value = 0
        notifyError(err)
      } finally {
        if (seed === requestSeed) loading.value = false
      }
    }

    async function loadAuthors() {
      authorsLoading.value = true
      authorsError.value = ''
      try {
        const data = await api.get('/api/public/authors/hot', { params: { limit: 6 } })
        authors.value = Array.isArray(data) ? data : []
      } catch (err) {
        authorsError.value = err.message || '热门作者加载失败'
      } finally {
        authorsLoading.value = false
      }
    }

    function changeSort(next) {
      if (sort.value === next) return
      sort.value = next
      page.value = 1
      loadNotes()
    }

    function changePage(next) {
      page.value = next
      loadNotes()
      const feed = document.getElementById('feed')
      if (feed) feed.scrollIntoView({ behavior: 'smooth', block: 'start' })
    }

    onMounted(() => {
      loadNotes()
      loadAuthors()
    })

    return {
      notes,
      total,
      page,
      size,
      sort,
      loading,
      error,
      authors,
      authorsLoading,
      authorsError,
      loggedIn,
      loadNotes,
      loadAuthors,
      changeSort,
      changePage
    }
  }
}
</script>
```

```vue
import { computed, onMounted, reactive, ref } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import AuthShell from '@/components/AuthShell.vue'

import { notifyError, notifySuccess } from '@/store/toast'
import { displayName } from '@/utils/format'

export default {
  name: 'LoginView',
  components: { AuthShell },
  setup() {
    const route = useRoute()
    const router = useRouter()
    const form = reactive({ username: '', password: '' })
    const loading = ref(false)
    const error = ref('')
    const showPassword = ref(false)
    const registered = computed(() => route.query.registered === '1')

    onMounted(() => {
      if (typeof route.query.username === 'string') {
        form.username = route.query.username
      }
      if (route.query.registered === '1') {
        notifySuccess('注册成功，请登录')
      }
    })

    function fillDemo(username, password) {
      form.username = username
      form.password = password
      error.value = ''
    }

    function resolveRedirect() {
      const redirect = route.query.redirect
      if (typeof redirect !== 'string' || !redirect.startsWith('/')) return '/'
      if (redirect.startsWith('/login') || redirect.startsWith('/register')) return '/'
      return redirect
    }

    async function submit() {
      error.value = ''
      if (!form.username.trim()) {
        error.value = '请输入用户名'
        return
      }
      if (!form.password) {
        error.value = '请输入密码'
        return
      }
      loading.value = true
      try {
        await auth.login(form.username.trim(), form.password)
        notifySuccess(`欢迎回来，${displayName(auth.state.user)}`)
        router.replace(resolveRedirect())
      } catch (err) {
        error.value = err.message || '登录失败'
        notifyError(err)
      } finally {
        loading.value = false
      }
    }

    return { form, loading, error, showPassword, registered, submit, fillDemo }
  }
}
```

```vue
<script>
import { onMounted, ref } from 'vue'
import api from '@/api'
import { notifyError, notifySuccess } from '@/store/toast'
import AppPagination from '@/components/AppPagination.vue'
import EmptyState from '@/components/EmptyState.vue'

const PAGE_SIZE = 8

  name: 'MyNotesView',

  setup() {
    const notes = ref([])
    const total = ref(0)
    const page = ref(1)
    const loading = ref(true)
    const error = ref('')
    const deletingId = ref(null)
    const size = PAGE_SIZE
    let requestSeed = 0

    async function loadNotes() {
      const seed = ++requestSeed
      loading.value = true
      error.value = ''
      try {
        const data = await api.get('/api/notes/mine', {
          params: { page: page.value, size: PAGE_SIZE }
        })
        if (seed !== requestSeed) return
        notes.value = Array.isArray(data.records) ? data.records : []
        total.value = Number(data.total) || 0
        const totalPages = Math.max(1, Math.ceil(total.value / PAGE_SIZE))
        if (page.value > totalPages) {
          page.value = totalPages
          loadNotes()
        }
      } catch (err) {
        if (seed !== requestSeed) return
        error.value = err.message || '我的笔记加载失败'
        notes.value = []
        total.value = 0
        notifyError(err)
      } finally {
        if (seed === requestSeed) loading.value = false
      }
    }

    function changePage(next) {
      page.value = next
      loadNotes()
      window.scrollTo({ top: 0, behavior: 'smooth' })
    }

    async function removeNote(note) {
      if (!note) return
      if (!window.confirm(`确定删除《${note.title || '无标题笔记'}》吗？删除后无法恢复。`)) return
      deletingId.value = note.id
      try {
        await api.delete(`/api/notes/${note.id}`)
        notifySuccess('笔记已删除')
        if (notes.value.length === 1 && page.value > 1) {
          page.value -= 1
        }
        await loadNotes()
      } catch (err) {
        notifyError(err)
      } finally {
        deletingId.value = null
      }
    }

    onMounted(loadNotes)

    return {
      notes,
      total,
      page,
      size,
      loading,
      error,
      deletingId,
      loadNotes,
      changePage,
      removeNote,
      excerpt,
      formatDate,
      statusMeta: noteStatusMeta
    }
  }
}
```

```vue
<script>
import { computed, onBeforeUnmount, onMounted, ref } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import api from '@/api'
import auth from '@/store/auth'
import { notifyError, notifySuccess } from '@/store/toast'
import UserAvatar from '@/components/UserAvatar.vue'
import EmptyState from '@/components/EmptyState.vue'
import {
  aiTypeLabel,
  countComments,
  flattenReplies,
  formatDate,
  mediaUrl,
  noteStatusMeta,
  timeAgo
} from '@/utils/format'

export default {
  name: 'NoteDetailView',
  components: { UserAvatar, EmptyState },
  setup() {
    const route = useRoute()
    const router = useRouter()

    const detail = ref(null)
    const loading = ref(true)
    const error = ref('')
    const author = ref(null)
    const deleting = ref(false)
    const liking = ref(false)

    const comments = ref([])
    const commentsLoading = ref(false)
    const commentsError = ref('')
    const commentContent = ref('')
    const submittingComment = ref(false)

    const replyTo = ref(null)
    const replyContent = ref('')
    const submittingReply = ref(false)

    const lightboxIndex = ref(null)

    const note = computed(() => (detail.value && detail.value.note) || {})
    const images = computed(() =>
      detail.value && Array.isArray(detail.value.images) ? detail.value.images : []
    )
    const liked = computed(() => Boolean(detail.value && detail.value.liked))
    const loggedIn = computed(() => auth.isLoggedIn())
    const currentUser = computed(() => auth.state.user || {})
    const owner = computed(
      () => Boolean(currentUser.value.id && note.value.userId === currentUser.value.id)
    )
    const canComment = computed(() => note.value.isPublic === 1 && note.value.status === 1)
    const canLike = computed(() => canComment.value)
    const statusMeta = computed(() => noteStatusMeta(note.value.status))
    const commentTotal = computed(() => countComments(comments.value))
    const commentTree = computed(() =>
      comments.value.map((root) => ({
        ...root,
        replies: flattenReplies(root.children)
      }))
    )
    const loginLink = computed(() => {
      const redirect = encodeURIComponent(route.fullPath)
      return `/login?redirect=${redirect}`
    })

    async function load() {
      loading.value = true
      error.value = ''
      detail.value = null
      author.value = null
      comments.value = []
      replyTo.value = null
      const id = route.params.id
      try {
        const endpoint = auth.isLoggedIn()
          ? `/api/notes/${id}`
          : `/api/public/notes/${id}`
        const data = await api.get(endpoint)
        detail.value = data
        if (data && data.note && data.note.userId) {
          loadAuthor(data.note.userId)
        }
        if (data && data.note && data.note.isPublic === 1) {
          loadComments()
        }
      } catch (err) {
        error.value = err.message || '笔记加载失败'
        notifyError(err)
      } finally {
        loading.value = false
      }
    }

    async function loadAuthor(userId) {
      try {
        author.value = await api.get(`/api/public/users/${userId}`)
      } catch (err) {
        author.value = null
      }
    }

    async function loadComments() {
      const id = route.params.id
      commentsLoading.value = true
      commentsError.value = ''
      try {
        const data = await api.get(`/api/public/notes/${id}/comments`)
        comments.value = Array.isArray(data) ? data : []
        if (detail.value && detail.value.note) {
          detail.value.note.commentCount = countComments(comments.value)
        }
      } catch (err) {
        commentsError.value = err.message || '评论加载失败'
      } finally {
        commentsLoading.value = false
      }
    }

    async function submitComment() {
      const content = commentContent.value.trim()
      if (!content) {
        notifyError('评论内容不能为空')
        return
      }
      if (content.length > 500) {
        notifyError('评论内容不能超过 500 个字符')
        return
      }
      submittingComment.value = true
      try {
        await api.post(`/api/notes/${route.params.id}/comments`, {
          content,
          parentId: null
        })
        commentContent.value = ''
        notifySuccess('评论发表成功')
        await loadComments()
      } catch (err) {
        notifyError(err)
      } finally {
        submittingComment.value = false
      }
    }

    function startReply(root, child) {
      if (isReplyingChild(root, child) || (!child && isReplyingRoot(root))) {
        cancelReply()
        return
      }
      replyTo.value = {
        rootId: root.id,
        targetId: child ? child.id : null,
        targetName: child ? child.nickname : root.nickname
      }
      replyContent.value = ''
    }

    function cancelReply() {
      replyTo.value = null
      replyContent.value = ''
    }

    function isReplyingRoot(root) {
      return Boolean(replyTo.value && replyTo.value.rootId === root.id && !replyTo.value.targetId)
    }

    function isReplyingChild(root, child) {
      return Boolean(
        replyTo.value && replyTo.value.rootId === root.id && replyTo.value.targetId === child.id
      )
    }

    async function submitReply(root) {
      const raw = replyContent.value.trim()
      if (!raw) {
        notifyError('回复内容不能为空')
        return
      }
      if (!replyTo.value) return
      const targetId = replyTo.value.targetId
      const targetName = replyTo.value.targetName
      const content = targetId ? `回复 @${targetName}：${raw}` : raw
      if (content.length > 500) {
        notifyError('回复内容不能超过 500 个字符')
        return
      }
      submittingReply.value = true
      try {
        await api.post(`/api/notes/${route.params.id}/comments`, {
          content,
          parentId: targetId || root.id
        })
        notifySuccess('回复成功')
        cancelReply()
        await loadComments()
      } catch (err) {
        notifyError(err)
      } finally {
        submittingReply.value = false
      }
    }

    function canDelete(comment) {
      if (!loggedIn.value || !comment) return false
      if (auth.isAdmin()) return true
      return comment.userId === currentUser.value.id
    }

    async function removeComment(comment) {
      if (!comment) return
      if (!window.confirm('确定删除这条评论吗？其下的回复也会一并删除。')) return
      try {
        if (auth.isAdmin()) {
          await api.delete(`/api/admin/comments/${comment.id}`)
        } else {
          await api.delete(`/api/comments/${comment.id}`)
        }
        notifySuccess('评论已删除')
        await loadComments()
      } catch (err) {
        notifyError(err)
      }
    }

    async function toggleLike() {
      if (!loggedIn.value) {
        router.push({ name: 'login', query: { redirect: route.fullPath } })
        return
      }
      liking.value = true
      try {
        const data = await api.post(`/api/notes/${route.params.id}/like`)
        if (detail.value) {
          detail.value.liked = Boolean(data && data.liked)
          if (detail.value.note) {
            detail.value.note.likeCount = Number(data && data.likeCount) || 0
          }
        }
        notifySuccess(detail.value && detail.value.liked ? '已点赞' : '已取消点赞')
      } catch (err) {
        notifyError(err)
      } finally {
        liking.value = false
      }
    }

    async function removeNote() {
      if (!detail.value) return
      if (!window.confirm('确定删除这篇笔记吗？删除后无法恢复。')) return
      deleting.value = true
      try {
        await api.delete(`/api/notes/${note.value.id}`)
        notifySuccess('笔记已删除')
        router.replace('/me')
      } catch (err) {
        notifyError(err)
      } finally {
        deleting.value = false
      }
    }

    function openLightbox(index) {
      lightboxIndex.value = index
    }

    function closeLightbox() {
      lightboxIndex.value = null
    }

    function stepLightbox(step) {
      if (lightboxIndex.value === null) return
      const length = images.value.length
      if (!length) return
      lightboxIndex.value = (lightboxIndex.value + step + length) % length
    }

    function handleKeydown(event) {
      if (lightboxIndex.value === null) return
      if (event.key === 'Escape') closeLightbox()
      if (event.key === 'ArrowLeft') stepLightbox(-1)
      if (event.key === 'ArrowRight') stepLightbox(1)
    }

    onMounted(() => {
      load()
      window.addEventListener('keydown', handleKeydown)
    })

    onBeforeUnmount(() => {
      window.removeEventListener('keydown', handleKeydown)
    })

    return {
      detail,
      loading,
      error,
      author,
      deleting,
      liking,
      comments,
      commentsLoading,
      commentsError,
      commentContent,
      submittingComment,
      replyTo,
      replyContent,
      submittingReply,
      lightboxIndex,
      note,
      images,
      liked,
      loggedIn,
      currentUser,
      owner,
      canComment,
      canLike,
      statusMeta,
      commentTotal,
      commentTree,
      loginLink,
      load,
      loadComments,
      submitComment,
      startReply,
      cancelReply,
      isReplyingRoot,
      isReplyingChild,
      submitReply,
      canDelete,
      removeComment,
      toggleLike,
      removeNote,
      openLightbox,
      closeLightbox,
      stepLightbox,
      aiTypeLabel,
      formatDate,
      timeAgo,
      mediaUrl
    }
  }
}
</script>
```

```vue
<script>
import { computed, onMounted, reactive, ref } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import api, { uploadFile } from '@/api'
import { notifyError, notifySuccess } from '@/store/toast'
import EmptyState from '@/components/EmptyState.vue'
import { mediaUrl } from '@/utils/format'

const MAX_IMAGES = 9
const MAX_FILE_SIZE = 10 * 1024 * 1024
const ALLOWED_TYPES = ['image/jpeg', 'image/png', 'image/gif', 'image/webp']

export default {
  name: 'NoteEditorView',
  components: { EmptyState },
  setup() {
    const route = useRoute()
    const router = useRouter()

    const isEdit = computed(() => route.name === 'note-edit')
    const detail = ref(null)
    const loadingNote = ref(isEdit.value)
    const loadError = ref('')
    const submitting = ref(false)
    const submitError = ref('')

    const uploading = ref(false)
    const uploadTotal = ref(0)
    const uploadDone = ref(0)
    const uploadError = ref('')

    const form = reactive({
      bookName: '',
      title: '',
      content: '',
      isPublic: 0,
      images: []
    })

    const rejectReason = computed(() => {
      if (!detail.value || !detail.value.note) return ''
      return detail.value.note.rejectReason || ''
    })

    onMounted(async () => {
      if (!isEdit.value) {
        loadingNote.value = false
        return
      }
      try {
        const data = await api.get(`/api/notes/${route.params.id}`)
        detail.value = data
        const note = (data && data.note) || {}
        form.bookName = note.bookName || ''
        form.title = note.title || ''
        form.content = note.content || ''
        form.isPublic = note.isPublic === 1 ? 1 : 0
        form.images = Array.isArray(data && data.images) ? [...data.images] : []
      } catch (err) {
        loadError.value = err.message || '笔记加载失败'
        notifyError(err)
      } finally {
        loadingNote.value = false
      }
    })

    function removeImage(index) {
      form.images.splice(index, 1)
    }

    async function onFilesChange(event) {
      const input = event.target
      let files = Array.from(input.files || [])
      input.value = ''
      if (!files.length) return

      uploadError.value = ''

      const slots = MAX_IMAGES - form.images.length
      if (slots <= 0) {
        uploadError.value = `最多只能上传 ${MAX_IMAGES} 张图片`
        notifyError(uploadError.value)
        return
      }
      if (files.length > slots) {
        notifyError(`最多还能上传 ${slots} 张，已自动截取前 ${slots} 张`)
        files = files.slice(0, slots)
      }

      const invalid = files.find(
        (file) => !ALLOWED_TYPES.includes(file.type) || file.size > MAX_FILE_SIZE
      )
      if (invalid) {
        uploadError.value = `“${invalid.name}”格式不支持或超过 10MB，仅支持 JPG / PNG / GIF / WebP`
        notifyError(uploadError.value)
        return
      }

      uploading.value = true
      uploadTotal.value = files.length
      uploadDone.value = 0
      try {
        for (const file of files) {
          const data = await uploadFile(file)
          if (!data || !data.url) {
            throw new Error('上传成功但未返回图片地址')
          }
          form.images.push(data.url)
          uploadDone.value += 1
        }
        notifySuccess(`成功上传 ${uploadDone.value} 张图片`)
      } catch (err) {
        uploadError.value = err.message || '图片上传失败'
        notifyError(err)
      } finally {
        uploading.value = false
        uploadTotal.value = 0
        uploadDone.value = 0
      }
    }

    function validate() {
      if (!form.bookName.trim()) return '请填写书名'
      if (form.bookName.trim().length > 100) return '书名不能超过 100 个字符'
      if (!form.title.trim()) return '请填写标题'
      if (form.title.trim().length > 100) return '标题不能超过 100 个字符'
      if (!form.content.trim()) return '正文不能为空'
      if (form.images.length > MAX_IMAGES) return `最多上传 ${MAX_IMAGES} 张图片`
      return ''
    }

    async function submit() {
      submitError.value = ''
      const message = validate()
      if (message) {
        submitError.value = message
        notifyError(message)
        return
      }

      submitting.value = true
      try {
        const payload = {
          bookName: form.bookName.trim(),
          title: form.title.trim(),
          content: form.content.trim(),
          isPublic: form.isPublic,
          images: [...form.images]
        }

        let data
        if (isEdit.value) {
          data = await api.put(`/api/notes/${route.params.id}`, payload)
        } else {
          data = await api.post('/api/notes', payload)
        }

        if (payload.isPublic === 1) {
          notifySuccess(
            isEdit.value ? '已保存，公开笔记已重新提交审核' : '已提交审核，审核通过后将在首页展示'
          )
        } else {
          notifySuccess(isEdit.value ? '笔记已保存' : '笔记已发布，仅自己可见')
        }

        const newId = data && data.note ? data.note.id : route.params.id
        if (!isEdit.value && payload.isPublic === 0 && newId) {
          router.replace(`/notes/${newId}`)
        } else {
          router.replace('/me')
        }
      } catch (err) {
        submitError.value = err.message || '保存失败，请稍后重试'
        notifyError(err)
      } finally {
        submitting.value = false
      }
    }

    return {
      isEdit,
      detail,
      loadingNote,
      loadError,
      submitting,
      submitError,
      uploading,
      uploadTotal,
      uploadDone,
      uploadError,
      form,
      rejectReason,
      removeImage,
      onFilesChange,
      submit,
      mediaUrl
    }
  }
}
</script>
```

```vue
<script>
import { computed, onMounted, reactive, ref } from 'vue'
import api, { uploadFile } from '@/api'
import auth from '@/store/auth'
import { notifyError, notifySuccess } from '@/store/toast'
import UserAvatar from '@/components/UserAvatar.vue'
import EmptyState from '@/components/EmptyState.vue'
import { formatDate } from '@/utils/format'

const ALLOWED_TYPES = ['image/jpeg', 'image/png', 'image/gif', 'image/webp']
const MAX_FILE_SIZE = 10 * 1024 * 1024

export default {
  name: 'ProfileView',
  components: { UserAvatar, EmptyState },
  setup() {
    const user = ref(null)
    const loading = ref(true)
    const error = ref('')
    const saving = ref(false)
    const saveError = ref('')
    const saved = ref(false)
    const savedAt = ref('')

    const uploadingAvatar = ref(false)

    const form = reactive({ nickname: '', bio: '', avatar: '' })
    const counts = reactive({ followers: 0, following: 0 })
    const activeList = ref('followers')
    const people = ref([])
    const peopleTotal = ref(0)
    const peopleLoading = ref(false)
    const peopleError = ref('')

    const isAdmin = computed(() => Boolean(user.value && user.value.role === 'ADMIN'))

    function fillForm(data) {
      form.nickname = data.nickname || ''
      form.bio = data.bio || ''
      form.avatar = data.avatar || ''
    }

    async function loadProfile() {
      loading.value = true
      error.value = ''
      try {
        const data = await api.get('/api/users/me')
        user.value = data
        fillForm(data)
        saved.value = false
        await Promise.all([loadCounts(data.id), loadPeople(activeList.value, data.id)])
      } catch (err) {
        error.value = err.message || '个人资料加载失败'
        notifyError(err)
      } finally {
        loading.value = false
      }
    }

    async function loadCounts(userId) {
      try {
        const [followers, following] = await Promise.all([
          api.get(`/api/users/${userId}/followers`, { params: { page: 1, size: 1 } }),
          api.get(`/api/users/${userId}/following`, { params: { page: 1, size: 1 } })
        ])
        counts.followers = Number(followers && followers.total) || 0
        counts.following = Number(following && following.total) || 0
      } catch (err) {
        notifyError(err)
      }
    }

    async function loadPeople(kind = activeList.value, userId = null) {
      const id = userId || (user.value && user.value.id)
      if (!id) return
      peopleLoading.value = true
      peopleError.value = ''
      try {
        const data = await api.get(`/api/users/${id}/${kind}`, {
          params: { page: 1, size: 10 }
        })
        people.value = Array.isArray(data.records) ? data.records : []
        peopleTotal.value = Number(data.total) || 0
      } catch (err) {
        peopleError.value = err.message || '列表加载失败'
      } finally {
        peopleLoading.value = false
      }
    }

    function selectList(kind) {
      if (kind === activeList.value && people.value.length) return
      activeList.value = kind
      loadPeople(kind)
    }

    function resetForm() {
      if (!user.value) return
      fillForm(user.value)
      saveError.value = ''
      saved.value = false
    }

    async function onAvatarChange(event) {
      const input = event.target
      const file = input.files && input.files[0]
      input.value = ''
      if (!file) return
      if (!ALLOWED_TYPES.includes(file.type) || file.size > MAX_FILE_SIZE) {
        const message = '头像仅支持 JPG / PNG / GIF / WebP，且不超过 10MB'
        saveError.value = message
        notifyError(message)
        return
      }
      uploadingAvatar.value = true
      saveError.value = ''
      try {
        const data = await uploadFile(file)
        if (!data || !data.url) throw new Error('上传成功但未返回图片地址')
        form.avatar = data.url
        notifySuccess('头像上传成功，点击「保存资料」后生效')
      } catch (err) {
        saveError.value = err.message || '头像上传失败'
        notifyError(err)
      } finally {
        uploadingAvatar.value = false
      }
    }

    async function saveProfile() {
      saveError.value = ''
      saved.value = false
      if (form.nickname.length > 32) {
        saveError.value = '昵称不能超过 32 个字符'
        return
      }
      if (form.bio.length > 200) {
        saveError.value = '简介不能超过 200 个字符'
        return
      }
      saving.value = true
      try {
        const data = await api.put('/api/users/me', {
          nickname: form.nickname.trim(),
          bio: form.bio.trim(),
          avatar: form.avatar.trim()
        })
        user.value = data
        fillForm(data)
        auth.setUser(data)
        saved.value = true
        savedAt.value = formatDate(new Date().toISOString())
        notifySuccess('个人资料已保存')
      } catch (err) {
        saveError.value = err.message || '保存失败，请稍后重试'
        notifyError(err)
      } finally {
        saving.value = false
      }
    }

    onMounted(loadProfile)

    return {
      user,
      loading,
      error,
      saving,
      saveError,
      saved,
      savedAt,
      uploadingAvatar,
      form,
      counts,
      activeList,
      people,
      peopleTotal,
      peopleLoading,
      peopleError,
      isAdmin,
      loadProfile,
      loadPeople,
      selectList,
      resetForm,
      onAvatarChange,
      saveProfile
    }
  }
}
</script>
```

```vue
<script>
import { reactive, ref } from 'vue'
import { useRouter } from 'vue-router'
import AuthShell from '@/components/AuthShell.vue'
import api from '@/api'
import { notifyError, notifySuccess } from '@/store/toast'

export default {
  name: 'RegisterView',
  components: { AuthShell },
  setup() {
    const router = useRouter()
    const form = reactive({
      username: '',
      nickname: '',
      password: '',
      confirmPassword: ''
    })
    const loading = ref(false)
    const error = ref('')

    function validate() {
      const username = form.username.trim()
      if (username.length < 4 || username.length > 32) {
        return '用户名长度需为 4–32 个字符'
      }
      if (form.nickname.trim().length > 32) {
        return '昵称不能超过 32 个字符'
      }
      if (form.password.length < 6 || form.password.length > 32) {
        return '密码长度需为 6–32 个字符'
      }
      if (form.password !== form.confirmPassword) {
        return '两次输入的密码不一致'
      }
      return ''
    }

    async function submit() {
      error.value = ''
      const message = validate()
      if (message) {
        error.value = message
        return
      }
      loading.value = true
      try {
        const payload = {
          username: form.username.trim(),
          password: form.password
        }
        if (form.nickname.trim()) payload.nickname = form.nickname.trim()
        await api.post('/api/auth/register', payload)
        notifySuccess('注册成功，请登录')
        router.replace({ name: 'login', query: { username: payload.username, registered: '1' } })
      } catch (err) {
        error.value = err.message || '注册失败，请稍后重试'
        notifyError(err)
      } finally {
        loading.value = false
      }
    }

    return { form, loading, error, submit }
  }
}
</script>
```

```vue
<script>
import { computed, onMounted, ref } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import api from '@/api'
import auth from '@/store/auth'
import { notifyError, notifySuccess } from '@/store/toast'
import UserAvatar from '@/components/UserAvatar.vue'
import NoteCard from '@/components/NoteCard.vue'
import AppPagination from '@/components/AppPagination.vue'
import EmptyState from '@/components/EmptyState.vue'
import { hydrateNoteCovers } from '@/utils/hydrate'

const PAGE_SIZE = 6

export default {
  name: 'UserProfileView',
  components: { UserAvatar, NoteCard, AppPagination, EmptyState },
  setup() {
    const route = useRoute()
    const router = useRouter()

    const profile = ref(null)
    const loading = ref(true)
    const error = ref('')
    const following = ref(false)
    const followBusy = ref(false)

    const notes = ref([])
    const total = ref(0)
    const page = ref(1)
    const notesLoading = ref(true)
    const notesError = ref('')
    const size = PAGE_SIZE
    let notesSeed = 0

    const loggedIn = computed(() => auth.isLoggedIn())
    const isSelf = computed(() => {
      const user = auth.state.user
      return Boolean(user && profile.value && user.id === profile.value.id)
    })

    async function loadProfile() {
      loading.value = true
      error.value = ''
      try {
        const data = await api.get(`/api/public/users/${route.params.id}`)
        profile.value = data
        following.value = Boolean(data.followed)
        if (loggedIn.value && !isSelf.value) {
          checkFollowing(data.id)
        }
      } catch (err) {
        error.value = err.message || '作者信息加载失败'
        notifyError(err)
      } finally {
        loading.value = false
      }
    }

    /**
     * 公开作者接口的 followed 字段依赖当前用户，但后端公开接口未传当前用户。
     * 这里通过「我关注的人」列表补一次关注状态，失败时忽略。
     */
    async function checkFollowing(targetId) {
      try {
        const data = await api.get(
          `/api/users/${auth.state.user.id}/following`,
          { params: { page: 1, size: 200 } }
        )
        const records = Array.isArray(data.records) ? data.records : []
        if (records.some((item) => item.id === targetId)) {
          following.value = true
        }
      } catch (err) {
        // 关注状态修正失败不阻塞页面
      }
    }

    async function loadNotes() {
      const seed = ++notesSeed
      notesLoading.value = true
      notesError.value = ''
      try {
        const data = await api.get(`/api/public/users/${route.params.id}/notes`, {
          params: { page: page.value, size: PAGE_SIZE }
        })
        if (seed !== notesSeed) return
        notes.value = Array.isArray(data.records) ? data.records : []
        total.value = Number(data.total) || 0
        hydrateNoteCovers(notes.value, () => seed !== notesSeed)
      } catch (err) {
        if (seed !== notesSeed) return
        notesError.value = err.message || '笔记列表加载失败'
        notes.value = []
        total.value = 0
      } finally {
        if (seed === notesSeed) notesLoading.value = false
      }
    }

    function changePage(next) {
      page.value = next
      loadNotes()
      window.scrollTo({ top: 0, behavior: 'smooth' })
    }

    async function toggleFollow() {
      if (!loggedIn.value) {
        router.push({ name: 'login', query: { redirect: route.fullPath } })
        return
      }
      if (!profile.value || isSelf.value) return
      followBusy.value = true
      try {
        const data = following.value

      } catch (err) {
        notifyError(err)
      } finally {
        followBusy.value = false
      }
    }

    onMounted(() => {
      loadProfile()
      loadNotes()
    })

    return {
      profile,
      loading,
      error,
      following,
      followBusy,

      changePage,
      toggleFollow
    }
  }
}
```
