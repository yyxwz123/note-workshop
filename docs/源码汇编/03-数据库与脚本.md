```javascript
// 用法：node scripts/api-test.mjs [baseUrl]

const BASE = process.argv[2] || 'http://127.0.0.1:8080';
const results = [];
const ctx = {};
const ts = Date.now().toString().slice(-6);

function expect(cond, msg) {
  if (!cond) throw new Error(msg);
}

async function call(method, path, { token, body, form, expectStatus } = {}) {
  const headers = {};
  if (token) headers.Authorization = `Bearer ${token}`;
  let payload;
  if (form) {
    payload = form;
  } else if (body !== undefined) {
    headers['Content-Type'] = 'application/json; charset=utf-8';
    payload = JSON.stringify(body);
  }
  const res = await fetch(BASE + path, { method, headers, body: payload });
  if (expectStatus && res.status !== expectStatus) {
    throw new Error(`HTTP ${res.status} != ${expectStatus}`);
  }
  const text = await res.text();
  let json = null;
  try { json = text ? JSON.parse(text) : null; } catch { /* 非 JSON（如静态文件） */ }
  return { status: res.status, json, text };
}

async function t(name, fn) {
  try {
    await fn();
    results.push({ name, ok: true });
    console.log(`  ✓ ${name}`);
  } catch (e) {
    results.push({ name, ok: false, detail: e.message });
    console.log(`  ✗ ${name}  -> ${e.message}`);
  }
}
```

```javascript
  for (const e of entries) {
```

```javascript
const browser = await puppeteer.launch({
  executablePath: EDGE,
  headless: 'new',
  args: ['--no-sandbox', '--lang=zh-CN'],
});
const page = await browser.newPage();
await page.setViewport({ width: 1440, height: 900 });
page.on('console', (m) => { if (m.type() === 'error') errors.push(m.text()); });
page.on('pageerror', (e) => errors.push(e.message));

let pass = true;
const check = (cond, msg) => {
  console.log(`${cond ? '✓' : '✗'} ${msg}`);
  if (!cond) pass = false;
};

try {
  // 1. 表单登录
  await page.goto(`${WEB}/login`, { waitUntil: 'networkidle2' });
  await page.type('#login-username', 'demo01');
  await page.type('#login-password', '123456');
  await Promise.all([
    page.waitForResponse((r) => r.url().includes('/api/auth/login'), { timeout: 15000 }),
    page.click('button[type="submit"]'),
  ]);
  await page.waitForFunction(() => location.pathname === '/', { timeout: 15000 });
  const token = await page.evaluate(() => localStorage.getItem('nw_token'));
  check(Boolean(token), '表单登录成功并跳转首页，token 已写入 localStorage');

  // 2. 发布公开笔记
  const title = `联调测试笔记 ${Date.now()}`;
  await page.goto(`${WEB}/notes/new`, { waitUntil: 'networkidle2' });
  await page.type('#note-book', '联调测试书目');
  await page.type('#note-title', title);
  await page.type('#note-content', '这是通过浏览器真实表单提交的联调测试笔记内容。');
  await page.click('input[value="1"]');
  const [resp] = await Promise.all([
    page.waitForResponse((r) => r.url().includes('/api/notes') && r.request().method() === 'POST', { timeout: 15000 }),
    page.click('button[type="submit"]'),
  ]);
  const body = await resp.json();
  check(body.code === 0, `发布接口返回 code=0（${body.message}）`);

  await new Promise((r) => setTimeout(r, 1500));
  await page.screenshot({ path: `${OUT}/11-publish-flow.png` });

  // 3. 校验数据：新笔记状态应为待审核(2)
  const mine = await page.evaluate(async (title) => {
    const token = localStorage.getItem('nw_token');
    const res = await fetch('/api/notes/mine?page=1&size=20', { headers: { Authorization: `Bearer ${token}` } });
    const body = await res.json();
    return body.data.records.find((n) => n.title === title) || null;
  }, title);
  check(Boolean(mine), '我的笔记接口能查到刚发布的笔记');
  check(mine && mine.status === 2, `新公开笔记状态为待审核（实际 ${mine && mine.status}）`);

  // 4. 校验公开列表不包含待审核笔记
  const inPublic = await page.evaluate(async (title) => {
    const res = await fetch('/api/public/notes?page=1&size=50');
    const body = await res.json();
    return body.data.records.some((n) => n.title === title);
  }, title);
  check(!inPublic, '待审核笔记未出现在公开列表');

  // 5. 退出登录
  await page.goto(`${WEB}/`, { waitUntil: 'networkidle2' });
  await page.evaluate(() => {
    localStorage.removeItem('nw_token');
    localStorage.removeItem('nw_user');
  });
  const guestToken = await page.evaluate(() => localStorage.getItem('nw_token'));
  check(guestToken === null, '清除登录态后 localStorage 已无 token');
} catch (e) {
  console.error('流程异常：', e.message);
  pass = false;
} finally {
  await browser.close();
}

if (errors.length) {
  console.log(`控制台错误 ${errors.length} 条：`, errors.slice(0, 3));
  pass = false;
}
console.log(pass ? '\n联调流程全部通过' : '\n联调流程存在失败项');
process.exitCode = pass ? 0 : 1;
```

```sql
-- 笔记工坊 Note-Workshop 全量建表脚本（V2）
-- MySQL 8.x

CREATE DATABASE IF NOT EXISTS note_workshop
  DEFAULT CHARACTER SET utf8mb4
  COLLATE utf8mb4_0900_ai_ci;

USE note_workshop;

-- 用户表
CREATE TABLE IF NOT EXISTS `user` (
  id          BIGINT       NOT NULL AUTO_INCREMENT COMMENT '主键',
  username    VARCHAR(32)  NOT NULL COMMENT '登录名',
  password    VARCHAR(100) NOT NULL COMMENT 'BCrypt 哈希',
  nickname    VARCHAR(32)  NOT NULL COMMENT '昵称',
  avatar      VARCHAR(255) DEFAULT NULL COMMENT '头像 URL',
  bio         VARCHAR(200) DEFAULT NULL COMMENT '个人简介',
  role        VARCHAR(16)  NOT NULL DEFAULT 'USER' COMMENT 'USER / ADMIN',
  status      TINYINT      NOT NULL DEFAULT 1 COMMENT '1 正常 0 禁用',
  created_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_user_username (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户';

-- 读书笔记表
CREATE TABLE IF NOT EXISTS note (
  id            BIGINT       NOT NULL AUTO_INCREMENT,
  user_id       BIGINT       NOT NULL COMMENT '作者',
  book_name     VARCHAR(100) NOT NULL COMMENT '图书名称',
  title         VARCHAR(100) NOT NULL COMMENT '笔记标题',
  content       TEXT         NOT NULL COMMENT '笔记正文',
  is_public     TINYINT      NOT NULL DEFAULT 0 COMMENT '1 公开 0 私有',
  status        TINYINT      NOT NULL DEFAULT 1 COMMENT '1 正常 2 待审核 3 审核驳回 0 已下架',
  reject_reason VARCHAR(200) DEFAULT NULL COMMENT '驳回原因',
  reviewed_at   DATETIME     DEFAULT NULL COMMENT '审核时间',
  like_count    INT          NOT NULL DEFAULT 0,
  comment_count INT          NOT NULL DEFAULT 0,
  created_at    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_note_user (user_id),
  KEY idx_note_public (is_public, status, like_count),
  KEY idx_note_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='读书笔记';

-- 笔记配图表
CREATE TABLE IF NOT EXISTS note_image (
  id       BIGINT       NOT NULL AUTO_INCREMENT,
  note_id  BIGINT       NOT NULL,
  url      VARCHAR(255) NOT NULL,
  sort     INT          NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  KEY idx_img_note (note_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='笔记配图';

-- 点赞表
CREATE TABLE IF NOT EXISTS note_like (
  id         BIGINT   NOT NULL AUTO_INCREMENT,
  user_id    BIGINT   NOT NULL,
  note_id    BIGINT   NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_like_user_note (user_id, note_id),
  KEY idx_like_note (note_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='点赞';

-- 评论表（支持二级回复）
CREATE TABLE IF NOT EXISTS comment (
  id         BIGINT       NOT NULL AUTO_INCREMENT,
  note_id    BIGINT       NOT NULL,
  user_id    BIGINT       NOT NULL,
  parent_id  BIGINT       NOT NULL DEFAULT 0 COMMENT '0 为一级评论',
  content    VARCHAR(500) NOT NULL,
  deleted    TINYINT      NOT NULL DEFAULT 0 COMMENT '逻辑删除',
  created_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_comment_note (note_id, deleted),
  KEY idx_comment_parent (parent_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='评论';

-- 关注表
CREATE TABLE IF NOT EXISTS follow (
  id          BIGINT   NOT NULL AUTO_INCREMENT,
  follower_id BIGINT   NOT NULL COMMENT '关注者',
  followee_id BIGINT   NOT NULL COMMENT '被关注者',
  created_at  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_follow (follower_id, followee_id),
  KEY idx_follow_followee (followee_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='关注';

-- AI 辅助结果（一篇笔记仅保留一条）
CREATE TABLE IF NOT EXISTS note_ai (
  id         BIGINT      NOT NULL AUTO_INCREMENT,
  note_id    BIGINT      NOT NULL,
  type       VARCHAR(20) NOT NULL COMMENT 'SUMMARY / AUDIENCE / SCORE',
  content    TEXT        NOT NULL,
  created_at DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_ai_note (note_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='笔记 AI 辅助结果';

-- AI 调用日志（统计成功/失败）
CREATE TABLE IF NOT EXISTS ai_call_log (
  id         BIGINT       NOT NULL AUTO_INCREMENT,
  user_id    BIGINT       DEFAULT NULL,
  scene      VARCHAR(32)  NOT NULL COMMENT 'SUMMARY/AUDIENCE/SCORE/CHAT',
  success    TINYINT      NOT NULL DEFAULT 0,
  error_msg  VARCHAR(255) DEFAULT NULL,
  latency_ms INT          NOT NULL DEFAULT 0,
  created_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_ai_log_time (created_at),
  KEY idx_ai_log_scene (scene)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='AI 调用日志';

-- AI 客服会话
CREATE TABLE IF NOT EXISTS ai_chat_session (
  id         BIGINT       NOT NULL AUTO_INCREMENT,
  user_id    BIGINT       NOT NULL,
  title      VARCHAR(100) NOT NULL DEFAULT '新的会话',
  created_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='AI 客服会话';

-- AI 客服消息
CREATE TABLE IF NOT EXISTS ai_chat_message (

  role       VARCHAR(16) NOT NULL COMMENT 'user / assistant',
  content    TEXT     NOT NULL,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_chat_msg_session (session_id, id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='AI 客服消息';
```
